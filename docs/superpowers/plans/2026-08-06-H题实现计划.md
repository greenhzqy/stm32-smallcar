# H题 — 车载平衡滚球运动控制系统 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在现有循迹小车基础上，新增 VL53L0X 球位置检测、MPU6050 姿态感知、舵机摆杆控制、串级 PID、6 模式任务调度，完整实现电赛 H 题。

**Architecture:** 5 个新模块（imu/ball/pendulum/chassis/task_manager），分层调用。硬件 I2C1 挂 VL53L0X+MPU6050，TIM1_CH1 驱动舵机，复用已有巡线 PID 和电机驱动。多频率任务调度（计数法）在 while(1) 中并行三组 PID。

**Tech Stack:** STM32F103C8T6 + HAL 库 + 硬件 I2C + TIM PWM + 纯 C

## Global Constraints

- **不修改已有代码逻辑**：motor.c、track.c、OLED.c 保持不动，只在 main.c 尾部追加新代码
- **GPIO 初始化在 `MX_GPIO_Init()` 中追加，TIM 初始化新增 `MX_TIM1_Init()`**
- **所有新增模块头文件使用 `__NAME_H` 格式的 include guard**
- **PID 参数使用 `float` 类型，带 `_f` 后缀**
- **I2C 地址传给 HAL 时必须 `<< 1`**
- **OLED 尺寸后续需换成 ≥2 英寸，当前 0.96 寸暂用于开发调试**
- **摆杆尚未搭建时，用 OLED 显示数值验证各模块功能**

```c
#include "imu.h"
#include <math.h>

static I2C_HandleTypeDef hi2c1_imu;  /* 模块内部持有 I2C1 */

void IMU_Init(void)
{
    /* 配置 I2C1 外设 */
    hi2c1_imu.Instance             = I2C1;
    hi2c1_imu.Init.ClockSpeed      = 400000;
    hi2c1_imu.Init.DutyCycle       = I2C_DUTYCYCLE_2;
    hi2c1_imu.Init.OwnAddress1     = 0;
    hi2c1_imu.Init.AddressingMode  = I2C_ADDRESSINGMODE_7BIT;
    hi2c1_imu.Init.DualAddressMode = I2C_DUALADDRESS_DISABLE;
    hi2c1_imu.Init.GeneralCallMode = I2C_GENERALCALL_DISABLE;
    hi2c1_imu.Init.NoStretchMode   = I2C_NOSTRETCH_DISABLE;
    HAL_I2C_Init(&hi2c1_imu);

    uint8_t data;
    /* 唤醒 MPU6050（写 0x00 到 PWR_MGMT_1 = 0x6B）*/
    data = 0x00;
    HAL_I2C_Mem_Write(&hi2c1_imu, MPU6050_ADDR << 1,
                      MPU6050_RA_PWR_MGMT_1, I2C_MEMADD_SIZE_8BIT,
                      &data, 1, 100);
    HAL_Delay(10);

    /* 加速度量程 ±2g（写 0x00 到 0x1C）*/
    data = 0x00;
    HAL_I2C_Mem_Write(&hi2c1_imu, MPU6050_ADDR << 1,
                      MPU6050_RA_ACCEL_CONFIG, I2C_MEMADD_SIZE_8BIT,
                      &data, 1, 100);

    /* 角速度量程 ±250°/s（写 0x00 到 0x1B）*/
    data = 0x00;
    HAL_I2C_Mem_Write(&hi2c1_imu, MPU6050_ADDR << 1,
                      MPU6050_RA_GYRO_CONFIG, I2C_MEMADD_SIZE_8BIT,
                      &data, 1, 100);
}

void IMU_Read(int16_t *ax, int16_t *ay, int16_t *az,
              int16_t *gx, int16_t *gy, int16_t *gz)
{
    uint8_t buf[14];
    HAL_I2C_Mem_Read(&hi2c1_imu, MPU6050_ADDR << 1,
                     MPU6050_RA_ACCEL_XOUT_H, I2C_MEMADD_SIZE_8BIT,
                     buf, 14, 100);

    *ax = (int16_t)((buf[0]  << 8) | buf[1]);
    *ay = (int16_t)((buf[2]  << 8) | buf[3]);
    *az = (int16_t)((buf[4]  << 8) | buf[5]);
    *gx = (int16_t)((buf[8]  << 8) | buf[9]);
    *gy = (int16_t)((buf[10] << 8) | buf[11]);
    *gz = (int16_t)((buf[12] << 8) | buf[13]);
}

float IMU_GetTilt(void)
{
    int16_t ax, ay, az, gx, gy, gz;
    IMU_Read(&ax, &ay, &az, &gx, &gy, &gz);
    /* 用 Y 和 Z 轴加速度算车身前后倾角（沿 X 轴的旋转）*/
    return atan2f((float)ay, (float)az) * 57.29578f;  /* 弧度转度 */
}

float IMU_GetYawRate(void)
{
    int16_t ax, ay, az, gx, gy, gz;
    IMU_Read(&ax, &ay, &az, &gx, &gy, &gz);
    /* MPU6050 ±250°/s 量程 → 原始值 ÷ 131 = °/s */
    return (float)gz / 131.0f;
}
```

**Notes:**

- 简化方案：不使用全功能 API，只写必要的初始化寄存器序列（约 50 行寄存器配置），然后用 `HAL_I2C_Mem_Read` 从 0x1E+0x1F 读距离

- 滤波：5 样本中值 + 相邻差 ≤ 20mm 限幅
- ball.h 需要 `extern I2C_HandleTypeDef hi2c1_imu;` 来复用 IMU 的 I2C 句柄，或者改成 public

> **🚧 设计决策：I2C 句柄共享**
> VL53L0X 和 MPU6050 共用 I2C1。为避免两个 .c 文件各自初始化 I2C1，采用两种方案之一：
> **方案 A（推荐）：** 在 imu.h 中 `extern I2C_HandleTypeDef hi2c1_imu;`，ball.c include imu.h 后直接使用。
> **方案 B：** 独立的 `i2c.c` 文件统一管理 I2C1，imu.c 和 ball.c 都 include。
> 本计划采用方案 A（少一个文件）。

- [ ] **Step 3: 创建 ball.c（初始化 + 读距离 + 滤波）**

```c
#include "ball.h"
#include "imu.h"   /* 复用 hi2c1_imu */

/* ================================================================
 * VL53L0X 初始化（简化序列，核心寄存器仅需约20行）
 *
 * 完整初始化要配上百个寄存器。这里用 ST 推荐的"最简序列"，
 * 对测距精度影响极小（±2mm），调试阶段完全够用。
 * ================================================================ */
void Ball_Init(void)
{
    uint8_t data;
    /* 等待 boot 完成 */
    HAL_Delay(2);

    /* 0x88=0x00, 0x80=0x01, 0xFF=0x01, 0x00=0x00 — 标准复位序列 */
    data = 0x00;
    HAL_I2C_Mem_Write(&hi2c1_imu, VL53L0X_ADDR << 1, 0x88,
                      I2C_MEMADD_SIZE_8BIT, &data, 1, 100);
    data = 0x01;
    HAL_I2C_Mem_Write(&hi2c1_imu, VL53L0X_ADDR << 1, 0x80,
                      I2C_MEMADD_SIZE_8BIT, &data, 1, 100);
    data = 0x01;
    HAL_I2C_Mem_Write(&hi2c1_imu, VL53L0X_ADDR << 1, 0xFF,
                      I2C_MEMADD_SIZE_8BIT, &data, 1, 100);
    data = 0x00;
    HAL_I2C_Mem_Write(&hi2c1_imu, VL53L0X_ADDR << 1, 0x00,
                      I2C_MEMADD_SIZE_8BIT, &data, 1, 100);

    /* 关闭标准模式（0x91=0x00: use long range）*/
    data = 0x00;
    HAL_I2C_Mem_Write(&hi2c1_imu, VL53L0X_ADDR << 1, 0x91,
                      I2C_MEMADD_SIZE_8BIT, &data, 1, 100);

    /* 信号速率限制 0.25Mcps */
    data = 0x00;
    HAL_I2C_Mem_Write(&hi2c1_imu, VL53L0X_ADDR << 1, 0x51,
                      I2C_MEMADD_SIZE_8BIT, &data, 1, 100);
    data = 0x00;
    HAL_I2C_Mem_Write(&hi2c1_imu, VL53L0X_ADDR << 1, 0x70,
                      I2C_MEMADD_SIZE_8BIT, &data, 1, 100);

    /* 预校准 VCSEL 周期 */
    data = 0x01;
    HAL_I2C_Mem_Write(&hi2c1_imu, VL53L0X_ADDR << 1, 0x61,
                      I2C_MEMADD_SIZE_8BIT, &data, 1, 100);

    /* 连续测距模式，通过写 0x00=0x02 启动 */
    HAL_Delay(100);
}

uint16_t Ball_ReadRaw(void)
{
    uint8_t buf[2];
    /* 读 0x1E（距离高字节）和 0x1F（低字节）*/
    HAL_I2C_Mem_Read(&hi2c1_imu, VL53L0X_ADDR << 1, 0x1E,
                     I2C_MEMADD_SIZE_8BIT, buf, 2, 100);
    return (uint16_t)((buf[0] << 8) | buf[1]);
}

/* ================================================================
 * 带滤波的读取
 *
 * 1. 中值滤波：缓存 5 个读数，排序取中间值
 * 2. 限幅滤波：与上一次有效值差 > 20mm → 丢弃（用旧值）
 * ================================================================ */
uint16_t Ball_GetDistance(void)
{
    static uint16_t ring[BALL_FILTER_SIZE] = {0};
    static uint8_t  idx = 0;
    static uint16_t last_valid = 100;  /* 初始值：假设球在 ~12.5cm */
    static uint8_t  ring_full = 0;

    uint16_t raw = Ball_ReadRaw();

    /* 限幅：相邻差 > 20mm 丢弃 */
    if (last_valid > 20) {
        int16_t diff = (int16_t)raw - (int16_t)last_valid;
        if (diff > BALL_LIMIT_DELTA || diff < -BALL_LIMIT_DELTA) {
            return last_valid;  /* 丢弃跳变值 */
        }
    }
    last_valid = raw;

    /* 写入环形缓冲区 */
    ring[idx] = raw;
    idx++;
    if (idx >= BALL_FILTER_SIZE) {
        idx = 0;
        ring_full = 1;
    }

    /* 中值排序 */
    uint16_t sorted[BALL_FILTER_SIZE];
    uint8_t n = ring_full ? BALL_FILTER_SIZE : idx;
    if (n == 0) return raw;
    for (uint8_t i = 0; i < n; i++) sorted[i] = ring[i];
    /* 简单冒泡（n≤5，开销可忽略）*/
    for (uint8_t i = 0; i < n - 1; i++)
        for (uint8_t j = i + 1; j < n; j++)
            if (sorted[j] < sorted[i]) {
                uint16_t t = sorted[i]; sorted[i] = sorted[j]; sorted[j] = t;
            }
    return sorted[n / 2];
}
```

{
    TIM_OC_InitTypeDef sConfigOC = {0};

    htim1.Init.Prescaler         = 71;      /* 72MHz÷72=1MHz, 1μs/步 */
    htim1.Init.CounterMode       = TIM_COUNTERMODE_UP;

    sConfigOC.OCPolarity = TIM_OCPOLARITY_HIGH;

}

**Notes:**

- CCR = 1500 + tilt × 11.1（1° ≈ 11.1 步，对应 ±45° 范围）
- 内环 PID 有积分限幅 ±200（防止舵机打死）
- 外环 PID 有积分限幅 ±3（管子的倾角不能太大）
- 管子倾角 = IMU_GetTilt() + 当前舵机命令角度（代码已知）

```c
#include "pendulum.h"
#include "imu.h"
#include "tim.h"

/* 内环 PID 参数 */
static float Kp_inner = 15.0f;   /* 倾角 PID — 待调 */
static float Ki_inner = 2.0f;
static float Kd_inner = 5.0f;
static float sum_inner  = 0.0f;
static float last_inner = 0.0f;

/* 外环 PID 参数 */
static float Kp_outer = 0.8f;    /* 球位置 PID — 待调 */
static float Ki_outer = 0.05f;
static float Kd_outer = 0.3f;
static float sum_outer  = 0.0f;
static float last_outer = 0.0f;

static float target_tilt_angle = 0.0f;  /* 外环输出 → 内环输入 */
static float servo_angle       = 0.0f;  /* 当前舵机命令角度（代码已知）*/
static float ball_target_mm    = 125.0f;/* 默认球目标 = 中心 12.5cm */

void Pendulum_Init(void)
{
    MX_TIM1_Init();  /* TIM1_CH1 PWM, CCR=1500 归中 */
    servo_angle = 0.0f;
}

void Pendulum_InnerLoop(float target_tilt)
{
    /* 获取管子实际倾角 = 车身倾角 + 舵机命令角度 */
    float body_tilt  = IMU_GetTilt();
    float tube_tilt  = body_tilt + servo_angle;  /* 近似 */
    float tilt_error = target_tilt - tube_tilt;

    /* PID */
    sum_inner += tilt_error;
    if (sum_inner > 200.0f)  sum_inner = 200.0f;
    if (sum_inner < -200.0f) sum_inner = -200.0f;

    float diff_inner = tilt_error - last_inner;
    float output = Kp_inner * tilt_error
                 + Ki_inner * sum_inner
                 + Kd_inner * diff_inner;

    /* output → 舵机角度 → CCR */
    servo_angle += output * 0.01f;   /* 积分到角度 */
    if (servo_angle > 40.0f)  servo_angle = 40.0f;
    if (servo_angle < -40.0f) servo_angle = -40.0f;

    uint16_t ccr = (uint16_t)(1500.0f + servo_angle * 11.1f);
    if (ccr < 500)  ccr = 500;
    if (ccr > 2500) ccr = 2500;
    __HAL_TIM_SET_COMPARE(&htim1, TIM_CHANNEL_1, ccr);

    last_inner = tilt_error;
}

void Pendulum_OuterLoop(float ball_target, float ball_actual)
{
    float ball_error = ball_target - ball_actual;

    sum_outer += ball_error;
    if (sum_outer > 3.0f)   sum_outer = 3.0f;
    if (sum_outer < -3.0f)  sum_outer = -3.0f;

    float diff_outer = ball_error - last_outer;
    float output = Kp_outer * ball_error
                 + Ki_outer * sum_outer
                 + Kd_outer * diff_outer;

    target_tilt_angle = output;  /* 更新外环输出 */
    last_outer = ball_error;
}

void Pendulum_SetTarget(float target_mm)
{
    ball_target_mm = target_mm;
}

float Pendulum_GetTargetTilt(void)
{
    return target_tilt_angle;
}
```

**Notes:**

- 长按 ≥ 2 秒 → 3 秒倒计时 → 开始执行
- 执行中短按 → 紧急停车
- 一圈完成 → 停车 → 显示结果 → 等 5 秒 → 回到待机

#define __TASK_MANAGER_H
#include "stm32f1xx_hal.h"

typedef enum {
    MODE_1 = 0,  /* 巡线一圈，停A点 */
    MODE_2,      /* 静止控球 */
    MODE_3,      /* A→B + 球中心 */
    MODE_4,      /* 一圈 + 球中心 */
    MODE_5,      /* 一圈 + 球指定位置 */
    MODE_6,      /* 预留 */
    MODE_COUNT
} TaskMode;

typedef enum {
    STATE_IDLE,       /* 待机，等待按键 */
    STATE_COUNTDOWN,  /* 3秒倒计时 */
    STATE_RUNNING,    /* 执行中 */
    STATE_DONE        /* 完成 */

void     TM_Update(void);

#endif

- [ ] **Step 2: 创建 task_manager.c**

```c
#include "task_manager.h"
#include "main.h"
#include "OLED.h"

static TaskMode  current_mode = MODE_1;
static TaskState state = STATE_IDLE;
static uint32_t  btn_press_start = 0;
static uint8_t   btn_prev = 1;     /* 上一次按键状态（1=未按）*/
static uint32_t  countdown_start  = 0;

void TM_Init(void)
{
    OLED_ShowString(0, 0, "Mode:1 Ready   ");
}

void TM_Update(void)
{
    uint8_t btn = HAL_GPIO_ReadPin(BUTTON_GPIO_Port, BUTTON_Pin);

    /* 下降沿检测（按下）*/
    uint8_t pressed = (btn_prev == 1 && btn == 0);
    uint8_t released = (btn_prev == 0 && btn == 1);
    btn_prev = btn;

    switch (state) {

    case STATE_IDLE:
        if (pressed) {
            btn_press_start = HAL_GetTick();
        }
        if (released) {
            uint32_t held = HAL_GetTick() - btn_press_start;
            if (held < 500) {
                /* 短按 → 切换模式 */
                current_mode = (TaskMode)((current_mode + 1) % MODE_COUNT);
                OLED_ShowNum(0, 5, current_mode + 1, 1);
            }
        }
        if (btn == 0 && (HAL_GetTick() - btn_press_start > 2000)) {
            /* 长按 2 秒 → 倒计时 */
            state = STATE_COUNTDOWN;
            countdown_start = HAL_GetTick();
        }
        break;

    case STATE_COUNTDOWN:
        {
            uint32_t elapsed = HAL_GetTick() - countdown_start;
            int remaining = 3 - (int)(elapsed / 1000);
            if (remaining >= 0) {
                OLED_ShowNum(2, 0, remaining, 1);
                OLED_ShowString(2, 2, "...");
            }
            if (elapsed >= 3000) {
                state = STATE_RUNNING;
                OLED_Clear();
                OLED_ShowString(0, 0, "Running...     ");
            }
        }
        break;

    case STATE_RUNNING:
        if (pressed) {
            /* 短按 → 紧急停车 */
            state = STATE_IDLE;
            OLED_Clear();
            OLED_ShowString(0, 0, "STOPPED        ");
        }
        break;

    case STATE_DONE:
        break;
    }
}

TaskMode TM_GetMode(void)  { return current_mode; }
uint8_t  TM_IsRunning(void) { return (state == STATE_RUNNING); }
```
